-- Ownership Filter Validation Loop
local function isHeldByPlayer(object)
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character and object:IsDescendantOf(player.Character) then
            return true
        end
        local backpack = player:FindFirstChildWhichIsA("Backpack")
        if backpack and object:IsDescendantOf(backpack) then
            return true
        end
    end
    return false
end

-- Core Render Mechanism
local function createUniversalESP(object, displayName, espColor, isAI)
    if ESP_TRACKING_CACHE[object] or object:FindFirstChild("UniversalItemESP") then return end
    ESP_TRACKING_CACHE[object] = true

    local targetPart
    if isAI then
        targetPart = object:WaitForChild("HumanoidRootPart", 3) or object:FindFirstChildWhichIsA("BasePart")
    else
        targetPart = object:IsA("BasePart") and object or object:FindFirstChildWhichIsA("BasePart")
    end
    
    if not targetPart then 
        ESP_TRACKING_CACHE[object] = nil
        return 
    end

    local espFolder = Instance.new("Folder")
    espFolder.Name = "UniversalItemESP"
    espFolder.Parent = object

    local highlight = Instance.new("Highlight")
    highlight.Adornee = object
    highlight.FillColor = espColor
    highlight.FillTransparency = isAI and 0.6 or 0.5
    highlight.OutlineColor = (espColor == Color3.fromRGB(0,0,0)) and Color3.fromRGB(255,255,255) or espColor -- High contrast outline for black engine part
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Parent = espFolder

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ItemLabel"
    billboard.Size = UDim2.new(0, 150, 0, 40)
    billboard.AlwaysOnTop = true
    billboard.ExtentsOffset = Vector3.new(0, isAI and 3 or 1.5, 0)
    billboard.MaxDistance = 10000
    billboard.Adornee = targetPart
    billboard.Parent = espFolder

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextColor3 = (espColor == Color3.fromRGB(0,0,0)) and Color3.fromRGB(255,255,255) or espColor -- Clear white text for black layout compatibility
    label.TextStrokeTransparency = 0
    label.TextStrokeColor3 = Color3.fromRGB(0,0,0)
    label.TextScaled = true
    label.Font = Enum.Font.SourceSansBold
    label.Parent = billboard

    task.spawn(function()
        local camera = workspace.CurrentCamera
        while object and object.Parent and espFolder.Parent do
            if not isAI and isHeldByPlayer(object) then
                espFolder:Destroy()
                break
            end

            if camera and targetPart and targetPart.Parent then
                local currentPosition = targetPart.Position
                if currentPosition.Magnitude < 1 and object:IsA("Model") then
                    currentPosition = object:GetPivot().Position
                end

                local distance = (camera.CFrame.Position - currentPosition).Magnitude
                
                if isAI then
                    label.Text = string.format("WARNING: %s\n[%d Studs]", string.upper(displayName), math.floor(distance))
                else
                    label.Text = string.format("%s\n[%d Studs]", string.upper(displayName), math.floor(distance))
                end
            else
                label.Text = ""
            end
            task.wait(isAI and 0.05 or 0.3)
        end
        ESP_TRACKING_CACHE[object] = nil
    end)
end

-- Sorting Engine
local function checkAndApply(desc)
    local lowerName = string.lower(desc.Name)
    if string.find(lowerName, "prop") then return end 
    
    -- 1. Main Enemy Threat AI Profiles (Granny, Grandpa, Slendrina)
    if lowerName == "granny" or lowerName == "grandpa" or lowerName == "slendrina" then
        if desc:IsA("Model") and (string.find(desc:GetFullName(), "Locks") or desc:FindFirstChild("Humanoid") or desc:FindFirstChild("HumanoidRootPart")) then
            local enemyColor = Color3.fromRGB(0, 255, 0) 
            if lowerName == "grandpa" then enemyColor = Color3.fromRGB(148, 0, 211) end
            if lowerName == "slendrina" then enemyColor = Color3.fromRGB(255, 105, 180) end
            
            createUniversalESP(desc, desc.Name, enemyColor, true)
            return
        end
    end

    -- 2. Robust Wildcard Size Check For Large Boss Spider
    if string.find(lowerName, "spider") and desc:IsA("Model") then
        if not string.find(lowerName, "key") and not string.find(lowerName, "room") and desc.Name ~= "Spiders" then
            if desc:FindFirstChildWhichIsA("BasePart") or desc:FindFirstChild("Humanoid") then
                local _, size = desc:GetPivot()
                -- Enforces volume limitation filter to drop tiny floor crawler clones completely
                if size and (size.X * size.Y * size.Z) > 15 then
                    createUniversalESP(desc, "BIG SPIDER", Color3.fromRGB(255, 69, 0), true)
                    return
                end
            end
        end
    end
    
    -- 3. Strict Structural Matrix Match For Loot & Vehicles
    local itemColor = ITEM_CONFIG[lowerName]
    if itemColor then
        if desc.Parent and string.lower(desc.Parent.Name) == lowerName then return end
        createUniversalESP(desc, desc.Name, itemColor, false)
    end
end

-- Execution Hooks
for _, desc in ipairs(workspace:GetDescendants()) do
    checkAndApply(desc)
end

workspace.DescendantAdded:Connect(function(desc)
    task.wait(0.4)
    checkAndApply(desc)
end)
